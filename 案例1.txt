def add_star(slide, left, top, size, color):
    """添加星星装饰 - 使用5角星"""
    try:
        star = slide.shapes.add_shape(MSO_AUTO_SHAPE_TYPE.STAR_5, left, top, size, size)
        star.fill.solid()
        star.fill.fore_color.rgb = color
        star.line.fill.background()
    except:
        # 如果星星不可用，用圆形替代
        star = slide.shapes.add_shape(MSO_SHAPE.OVAL, left, top, size, size)
        star.fill.solid()
        star.fill.fore_color.rgb = color
        star.line.fill.background()
    return star

def add_cloud(slide, left, top, width, height, color):
    """添加云朵装饰"""
    try:
        cloud = slide.shapes.add_shape(MSO_AUTO_SHAPE_TYPE.CLOUD, left, top, width, height)
        cloud.fill.solid()
        cloud.fill.fore_color.rgb = color
        cloud.line.fill.background()
    except:
        # 如果云朵不可用，用圆角矩形替代
        cloud = slide.shapes.add_shape(MSO_SHAPE.ROUNDED_RECTANGLE, left, top, width, height)
        cloud.fill.solid()
        cloud.fill.fore_color.rgb = color
        cloud.line.fill.background()
    return cloud

def add_smile_face(slide, left, top, size, color):
    """添加笑脸"""
    # 圆形脸
    face = slide.shapes.add_shape(MSO_SHAPE.OVAL, left, top, size, size)
    face.fill.solid()
    face.fill.fore_color.rgb = color
    face.line.fill.background()
    
    # 眼睛（两个小圆）
    eye1 = slide.shapes.add_shape(MSO_SHAPE.OVAL, left + size*0.2, top + size*0.25, size*0.12, size*0.15)
    eye1.fill.solid()
    eye1.fill.fore_color.rgb = RGBColor(50, 50, 50)
    eye1.line.fill.background()
    
    eye2 = slide.shapes.add_shape(MSO_SHAPE.OVAL, left + size*0.68, top + size*0.25, size*0.12, size*0.15)
    eye2.fill.solid()
    eye2.fill.fore_color.rgb = RGBColor(50, 50, 50)
    eye2.line.fill.background()
    
    # 嘴巴（弧形）
    mouth = slide.shapes.add_shape(MSO_SHAPE.ARC, left + size*0.25, top + size*0.45, size*0.5, size*0.35)
    mouth.fill.solid()
    mouth.fill.fore_color.rgb = RGBColor(200, 50, 50)
    mouth.line.fill.background()
    return face
