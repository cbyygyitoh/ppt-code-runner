# ========== 辅助函数 ==========
def add_bg(slide, color):
    bg = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, 0, 0, prs.slide_width, prs.slide_height)
    bg.fill.solid()
    bg.fill.fore_color.rgb = color
    bg.line.fill.background()

def add_text(slide, text, left, top, width, height, font_size=24, color=RGBColor(0,0,0), align=PP_ALIGN.CENTER, bold=False, italic=False):
    box = slide.shapes.add_textbox(left, top, width, height)
    tf = box.text_frame
    tf.text = text
    tf.paragraphs[0].alignment = align
    tf.paragraphs[0].font.size = Pt(font_size)
    tf.paragraphs[0].font.color.rgb = color
    tf.paragraphs[0].font.bold = bold
    tf.paragraphs[0].font.italic = italic
    return box

def add_deco_circle(slide, x, y, size, color):
    circle = slide.shapes.add_shape(MSO_SHAPE.OVAL, x, y, size, size)
    circle.fill.solid()
    circle.fill.fore_color.rgb = color
    circle.line.fill.background()
    return circle

def add_deco_line(slide, x, y, width, height, color):
    line = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, x, y, width, height)
    line.fill.solid()
    line.fill.fore_color.rgb = color
    line.line.fill.background()
    return line

def add_star(slide, x, y, size, color):
    star = slide.shapes.add_shape(MSO_SHAPE.STAR_5_POINT, x, y, size, size)
    star.fill.solid()
    star.fill.fore_color.rgb = color
    star.line.fill.background()
    return star
